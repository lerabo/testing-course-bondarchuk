```text
"""Академічна довідка студента (вимога Р-28).

Перенесено з системи деканату попередньої версії (2019) майже без змін.
"""

from student_portal.people import Funding

# кредити ЄКТС; інші дисципліни - 3 кредити
CREDITS = {"Python": 5, "Бази даних": 4, "Алгоритми": 5, "Веброзробка": 4}
W = 50


def transcript(student, group_name=""):  # noqa: C901 - технічний борг, див. Р-28
    s = ""
    s = s + "АКАДЕМІЧНА ДОВІДКА\n"
    s = s + "Студент: " + student.last_name + " " + student.first_name
    s = s + " (id " + str(student.id) + ")"
    if group_name != "":
        s = s + ", група " + group_name
    s = s + "\n"
    f = "контракт"
    if student.funding == Funding.BUDGET:
        f = "державне замовлення"
    s = s + "Форма навчання: " + f + "\n"
    if len(student.grades) == 0:
        s = s + "Оцінок немає\n"
        return s
    s = s + "-" * W + "\n"
    s = s + "Дисципліна".ljust(22) + "Кредити".rjust(8) + "Бали".rjust(6)
    s = s + "ECTS".rjust(6) + "  Оцінка\n"
    courses = []
    for g in student.grades:
        if g.course not in courses:
            courses.append(g.course)
    courses.sort()
    total = 0
    cr = 0
    debts = []
    for c in courses:
        # зараховується остання оцінка з дисципліни
        last = None
        for g in student.grades:
            if g.course == c and (last is None or g.graded_on >= last.graded_on):
                last = g
        k = CREDITS.get(c, 3)
        p = last.score
        if p >= 90:
            e = "A"
        elif p >= 82:
            e = "B"
        elif p >= 75:
            e = "C"
        elif p >= 67:
            e = "D"
        elif p >= 60:
            e = "E"
        elif p >= 35:
            e = "FX"
        else:
            e = "F"
        if p >= 90:
            n = "відмінно"
        elif p >= 74:
            n = "добре"
        elif p >= 60:
            n = "задовільно"
        else:
            n = "незадовільно"
            debts.append(c)
        total = total + p * k
        cr = cr + k
        s = s + c.ljust(22) + str(k).rjust(8) + str(p).rjust(6) + e.rjust(6)
        s = s + "  " + n + "\n"
    s = s + "-" * W + "\n"
    s = s + "Усього кредитів: " + str(cr) + "\n"
    avg = int(total / cr * 10 + 0.5) / 10
    s = s + "Середній бал (зважений за кредитами): " + str(avg).replace(".", ",")
    s = s + "\n"
    if len(debts) > 0:
        s = s + "Борги: " + ", ".join(debts) + "\n"
    else:
        s = s + "Боргів немає\n"
    return s

```

## Оцінювання

Покриття і мутаційний аналіз - на старій версії модуля (app-lab11) у тимчасовій гілці (`only_mutate = ["src/student_portal/transcript.py"]`, `also_copy = ["data/"]`, mutmut 3.8, 256 мутантів), по одному запуску на набір. mutmut запускається лише на зелених тестах, тож для набору ШІ «за специфікацією» три тести, що падають через #92 і #93, у гілці оцінювання тимчасово позначено `xfail(strict=True)` через `conftest.py` - сам файл не змінювався.

|---|---|---|---|---|---|---|
| Характеризаційний (мій) | 34 знімки | 34 / 0 | 100 % / 100 % (70 / 36) | 255 / 256 (99,6 %) | 34 / 0 | 24 / 10 до `--snapshot-update` - очікувано: знімок фіксував дефекти |

| ШІ «за специфікацією» | 41 | 38 / 3 | 100 % / 100 % | 247 / 256 (96,5 %) | 38 / 3 | 41 / 0 |

**Покриття нічого не розрізнило:** кожен набір проходить кожен рядок і кожну гілку модуля на 70 операторів і 36 розгалужень. Покриття каже лише, що код виконався, а не що результат перевірено правильно.

**Мутаційний показник розрізнив, але не правильність.** Найвищий показник (100 %) у набору, що закріплює обидва дефекти: мутанти `p >= 74` → `p > 74` чи `p >= 75` він «вбиває», бо очікує «добре» для 74, тобто саме дефект. Мутаційний аналіз вимірює, наскільки щільно тести тримають поточну поведінку, а оракулом для набору «за кодом» був сам код - це і є проблема оракула.

**Живі мутанти.**
- Характеризаційний: 1 - `group_name=""` → `"XXXX"` за замовчуванням. Мої знімки завжди передають групу явно, навіть порожню; виклику без аргументу немає. Набір ШІ «за кодом» його вбив (`test_empty_group_name_explicitly_is_same_as_default`).
- Мій за специфікацією: 6 - межа `>= 35` (`> 35`, `>= 36`) і рядки `"FX"`, `"F"` (`"fx"`, `"XXFXXX"`, `"f"`, `"XXFXX"`). Це код дефекту #92: за Р-28 таких літер немає, і їх перевіряють лише тести `xfail`, які падають на будь-якому мутанті. Після виправлення цього коду немає.
- ШІ «за специфікацією»: 9 - ті самі шість, `"F"` → `None` і `"незадовільно"` → `"XXнезадовільноXX"`, `"НЕЗАДОВІЛЬНО"`. Слово «незадовільно» в рядку таблиці набір перевіряє лише для балів 59 і 0, а ці тести падають через #92; тести боргів дивляться на рядок «Борги:», а не на оцінку в таблиці. Бракує граничних значень 35 і 34 і перевірки слова «незадовільно» на тесті, що проходить.

## Висновок

Прийняти варто весь набір «за специфікацією» (41 тест) і 48 тестів «за кодом» після зміни імпорту; відкинути - 7 тестів, що закріплюють дефекти, і 2, що перевіряють внутрішні константи. Два промпти дали протилежні результати за однакового покриття. Набір «за кодом» має найвищий мутаційний показник, але закріпив обидва дефекти: модель вивела очікування з коду, який тестувала, і навіть помітивши розбіжність шкал, оформила її як «очікувану поведінку». До того ж він прив'язаний до внутрішніх назв і зламався від першого ж перейменування під час рефакторингу. Набір «за специфікацією» знайшов обидва дефекти, пережив рефакторинг без змін і після виправлення став повністю зеленим, але пропустив межі 35/34 і має на три живі мутанти більше. Тести ШІ «за кодом» корисні як швидкий характеризаційний набір перед рефакторингом, а не як перевірка правильності: оракул для тесту має братися з вимоги, а не з коду.
